git branch - xem các nhánh và nhánh hiện tại
git checkout -b tennhanh  -> tạo và chuyển sang nhánh mới
git checkout tennhanh -> chuyển sang nhánh tennhanh
git add .  -> thêm tất cả thay đổi
git commit -m "vvv" -> đặt tên commit
git push -u origin tenhanh -> đấy thay đổi lên github vs thiết lập liên kết
git push -> dùng các lần khác sau khi dùng lệnh trên
git pull -> kéo thay đổi về máy
git merge tennhanh -> gộp nhánh tennhanh vào nhánh hiện tại
